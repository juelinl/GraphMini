set(standalone_headers
    backend/bit_ops/from_sorted.h
    common/types.h common/counter.h common/timer.h common/logging.h
    common/meta.h common/constant.h common/logitem.h
    compiler/config.h compiler/compilation_profile.h compiler/codegen.h
    compiler/ir.h compiler/execution_ir.h compiler/representation.h compiler/planning.h
    runtime/graph_loader.h runtime/graph_builder.h runtime/plan_module.h)
set(header_check_sources)
foreach(header IN LISTS standalone_headers)
    string(REPLACE "/" "_" header_name "${header}")
    set(source "${CMAKE_CURRENT_BINARY_DIR}/headers/${header_name}.cpp")
    file(GENERATE OUTPUT "${source}" CONTENT "#include \"${header}\"\n")
    list(APPEND header_check_sources "${source}")
endforeach()
add_library(header_self_containment OBJECT ${header_check_sources})
target_link_libraries(header_self_containment PRIVATE graphmini_backend)
