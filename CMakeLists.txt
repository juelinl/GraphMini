cmake_minimum_required(VERSION 3.20)
project(GraphMini)
option(GRAPHMINI_BUILD_TESTS "Build compiler regression tests" OFF)
option(GRAPHMINI_EXPERIMENTAL_NO_INLINE "Disable optional inlining in dynamic query compilation only" OFF)
option(GRAPHMINI_PROFILE_QUERY_COMPILATION "Record Clang time traces for dynamic query compilation" OFF)
if(GRAPHMINI_PROFILE_QUERY_COMPILATION AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Query compilation time traces require upstream Clang")
endif()
option(GRAPHMINI_EXPERIMENTAL_HEADER_UNITS "Use a Clang C++20 header unit for dynamic query plans" OFF)
option(GRAPHMINI_EXPERIMENTAL_TBB_MODULE "Build the official TBB module with a build-local compatibility fix" OFF)
option(GRAPHMINI_EXPERIMENTAL_BACKEND_MODULE "Import the complete GraphMini backend in dynamic queries" OFF)
if(GRAPHMINI_EXPERIMENTAL_BACKEND_MODULE)
    set(GRAPHMINI_EXPERIMENTAL_TBB_MODULE ON)
endif()
if(GRAPHMINI_EXPERIMENTAL_HEADER_UNITS AND GRAPHMINI_EXPERIMENTAL_TBB_MODULE)
    message(FATAL_ERROR "Select either header units or the named TBB module, not both")
endif()
if(GRAPHMINI_EXPERIMENTAL_HEADER_UNITS)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_GENERATOR MATCHES "Ninja")
        message(FATAL_ERROR "Experimental header units require upstream Clang and Ninja")
    endif()
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
endif()
if(GRAPHMINI_BUILD_TESTS)
    enable_testing()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
if(MSVC)
    set(CMAKE_CXX_FLAGS_RELEASE "$ENV{CXXFLAGS} /O2 /W3")
    set(CMAKE_CXX_FLAGS_DEBUG "$ENV{CXXFLAGS} /Od /W3 /Zi")
else()
    set(CMAKE_CXX_FLAGS_RELEASE "$ENV{CXXFLAGS} -O3 -Wall")
    set(CMAKE_CXX_FLAGS_DEBUG "$ENV{CXXFLAGS} -O0 -Wall -g -ggdb -fno-omit-frame-pointer")
endif()
# Prefer packages installed in the active Conda environment.
if(DEFINED ENV{CONDA_PREFIX})
    list(PREPEND CMAKE_PREFIX_PATH "$ENV{CONDA_PREFIX}")
    if(NOT OpenMP_ROOT)
        set(OpenMP_ROOT "$ENV{CONDA_PREFIX}")
    endif()
endif()
if(APPLE)
    set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -mno-omit-leaf-frame-pointer")
    if(NOT OpenMP_ROOT)
        execute_process(
                COMMAND brew --prefix libomp
                OUTPUT_VARIABLE OPENMP_ROOT_HINT
                OUTPUT_STRIP_TRAILING_WHITESPACE
                ERROR_QUIET)
        if(OPENMP_ROOT_HINT)
            set(OpenMP_ROOT "${OPENMP_ROOT_HINT}" CACHE PATH "Path to libomp" FORCE)
        endif()
    endif()
endif()
set(PYBIND11_FINDPYTHON ON)
set(Python3_FIND_VIRTUALENV FIRST)
if(NOT Python3_EXECUTABLE)
    if(DEFINED ENV{VIRTUAL_ENV})
        find_program(Python3_EXECUTABLE NAMES python python3
            PATHS "$ENV{VIRTUAL_ENV}/bin" "$ENV{VIRTUAL_ENV}/Scripts" NO_DEFAULT_PATH)
    elseif(DEFINED ENV{CONDA_PREFIX})
        find_program(Python3_EXECUTABLE NAMES python python3
            PATHS "$ENV{CONDA_PREFIX}/bin" "$ENV{CONDA_PREFIX}" NO_DEFAULT_PATH)
    endif()
endif()
find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
find_package(cxxopts CONFIG REQUIRED)
# Prefer the project-local source build when package channels lag upstream.
if(EXISTS "${PROJECT_SOURCE_DIR}/.deps/oneTBB-2023.1.0/lib/cmake/TBB/TBBConfig.cmake")
    list(PREPEND CMAKE_PREFIX_PATH "${PROJECT_SOURCE_DIR}/.deps/oneTBB-2023.1.0")
endif()
find_package(TBB 2023.1 CONFIG REQUIRED COMPONENTS tbb tbbmalloc)
message(STATUS "oneTBB ${TBB_VERSION}: ${TBB_DIR}")
if(GRAPHMINI_EXPERIMENTAL_TBB_MODULE)
    include(cmake/TbbModule.cmake)
endif()
find_package(fmt CONFIG REQUIRED)
find_package(pybind11 CONFIG REQUIRED)

find_package(OpenMP REQUIRED COMPONENTS CXX)

set(PROJECT_PROFILE_DIR "${PROJECT_SOURCE_DIR}/profile/")
set(PROJECT_LOG_DIR "${PROJECT_SOURCE_DIR}/log/")
set(PROJECT_PLAN_DIR "${PROJECT_SOURCE_DIR}/plan/")

# Generated text alone does not identify a usable query library: runtime headers
# and compiler/build settings also affect its ABI and behavior.
file(GLOB plan_runtime_inputs CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/src/backend/*.h" "${PROJECT_SOURCE_DIR}/include/*.h"
    "${PROJECT_SOURCE_DIR}/src/backend/set_ops/*.h"
    "${PROJECT_SOURCE_DIR}/src/backend/bit_ops/*.h"
    "${PROJECT_SOURCE_DIR}/src/codegen_output/plan.h")
list(APPEND plan_runtime_inputs "${PROJECT_SOURCE_DIR}/src/backend/graphmini_backend.cppm")
set(plan_build_fingerprint "${CMAKE_CXX_COMPILER};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_BUILD_TYPE};${CMAKE_CXX_FLAGS};${CMAKE_CXX_FLAGS_RELEASE};${CMAKE_CXX_FLAGS_DEBUG};${OpenMP_CXX_FLAGS};${TBB_VERSION};${TBB_DIR};${GRAPHMINI_EXPERIMENTAL_HEADER_UNITS};${GRAPHMINI_EXPERIMENTAL_TBB_MODULE}")
string(APPEND plan_build_fingerprint ";${GRAPHMINI_EXPERIMENTAL_BACKEND_MODULE}")
string(APPEND plan_build_fingerprint ";${GRAPHMINI_PROFILE_QUERY_COMPILATION}")
string(APPEND plan_build_fingerprint ";${GRAPHMINI_EXPERIMENTAL_NO_INLINE}")
foreach(input IN LISTS plan_runtime_inputs)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
    file(SHA256 "${input}" input_hash)
    string(APPEND plan_build_fingerprint ";${input_hash}")
endforeach()
string(SHA256 GRAPHMINI_PLAN_BUILD_ID "${plan_build_fingerprint}")

configure_file(include/configure.h.in ${PROJECT_BINARY_DIR}/configure/configure.h)
message("PROJECT_BINARY_DIR: " ${PROJECT_BINARY_DIR})
message("PROJECT_SOURCE_DIR: " ${PROJECT_SOURCE_DIR})
message("CMAKE_CXX_FLAGS_DEBUG: " ${CMAKE_CXX_FLAGS_DEBUG})
message("CMAKE_CXX_FLAGS_RELEASE: " ${CMAKE_CXX_FLAGS_RELEASE})
include_directories(${PROJECT_BINARY_DIR}/configure)
include_directories(include)
include_directories(src)
add_subdirectory(src)
